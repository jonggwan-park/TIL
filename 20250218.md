```python
import os
import streamlit as st
from dotenv import load_dotenv
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Pinecone as PineconeVectorStore
from langchain.chat_models import ChatOpenAI
from langchain.schema import HumanMessage
import pinecone

load_dotenv()
PINECONE_API_KEY = os.getenv("PINECONE_API_KEY")
PINECONE_ENV = os.getenv("PINECONE_ENV")
OPENAI_API_KEY = os.getenv("OPENAI_API_KEY")

# 1) Pinecone init (2.2.x)
pinecone.init(api_key=PINECONE_API_KEY, environment=PINECONE_ENV)
INDEX_NAME = "interview-index"

# 2) LangChain의 Pinecone VectorStore
vectorstore = PineconeVectorStore.from_existing_index(
    index_name=INDEX_NAME,
    embedding=OpenAIEmbeddings(openai_api_key=OPENAI_API_KEY),
    text_key="text"  # 메타데이터를 어느 필드에 담을지 (필요 시)
)

# 3) OpenAI LLM 설정
llm = ChatOpenAI(model="gpt-4", openai_api_key=OPENAI_API_KEY)

# 4) Streamlit App
st.title("🤖 AI 면접관 챗봇")
st.write("Pinecone + LangChain을 이용해 면접 질문을 자동으로 생성/검색해주는 챗봇")

user_input = st.text_input("면접관에게 질문하거나, 면접 상황을 입력해보세요:")

if st.button("질문하기"):
    if user_input:
        # 1) Pinecone에서 관련 면접 질문/해설 검색
        docs = vectorstore.similarity_search(user_input, k=2)

        # 2) 검색된 문서(컨텍스트) 추출
        context = "\n".join([doc.page_content for doc in docs])

        # 3) LLM 프롬프트 구성
        prompt = f"""
        아래는 면접 질문과 해설이 있습니다:
        {context}

        사용자의 요청: {user_input}

        - 당신은 AI 면접관입니다.
        - 지원자가 위와 같은 질문/답변을 했을 때, 
          가장 관련 있는 추가 면접 질문을 1~2개 더 제시해주세요.
        - 필요하다면 제공된 해설 기반으로 간단한 코멘트나 피드백도 해주세요.
        """

        # 4) GPT-4 호출
        response = llm([HumanMessage(content=prompt)])

        # 5) 결과 출력
        st.subheader("면접관 챗봇의 응답")
        st.write(response.content)
```

Pinecone 클라이언트가 올바른 인덱스에 연결하지 못했을 때 발생하는 오류라고 한다.

```python
import os
import json
import pinecone
from sentence_transformers import SentenceTransformer
from dotenv import load_dotenv

load_dotenv()

PINECONE_API_KEY = os.getenv("PINECONE_API_KEY")
PINECONE_ENV = os.getenv("PINECONE_ENV")

# pinecone 초기화
pinecone.init(api_key=PINECONE_API_KEY,environment=PINECONE_ENV)

INDEX_NAME = "interview-index" # 인덱스 이름 설정

# 인덱스 없으면 생성

if INDEX_NAME not in pinecone.list_indexes():
    pinecone.create_index(
        name=INDEX_NAME,
        dimension=384, # 아래 모델 차원에 맞춤
        metric="cosine"
    )

index = pinecone.Index(index_name=INDEX_NAME)

# 임베딩 모델 로드
model = SentenceTransformer("sentence-transformers/all-MiniLM-L6-v2")

def main():
    # json 파일 로드
    with open("data/interview_data.json","r",encoding="utf-8") as f:
        data_list = json.load(f)

    # pinecone에 업로드할 벡터 준비
    vectors_to_upsert = []
    for item in data_list:
        question_text = item["question"] + " " + item.get("explanation","")
        embedding = model.encode(question_text).tolist()

        metadata = {
            "question" :item["question"],
            "category" :item["category"],
            "difficulty":item["difficulty"],
            "explanation":item["explanation"],
            "tags":item["tags"]
        }
        vectors_to_upsert.append(
            (item["id"],embedding,metadata)
        )

    # pinecone에 업서트
    index.upsert(vectors=vectors_to_upsert)
    print("Data upserted to Pinecone successfully!")
    print(index.describe_index_stats())
    print(pinecone.list_indexes())  # 존재하는 인덱스 리스트 확인

if __name__ == "__main__":
    main()
```
챗 지피티의 도움을 받아 연습해보려고 했으나 계속 오류가 나는 관계로 시간을 많이 낭비했다.
